# Patterns for scripts/check-writing.sh. The check skips this file.
#
# Each line has the form:   SCOPE@@LABEL@@REGEX[@@EXTRA EXCLUDED PATHS]
#
#   SCOPE   tree    scan tracked files only
#           commit  scan commit messages only
#           both    scan tracked files and commit messages
#   REGEX   a PCRE, as git grep -P reads it
#   EXTRA   optional pathspecs, separated by spaces, that this pattern skips
#
# Lines that start with # and empty lines are ignored.
#
# A banned phrase right after a double quote is a mention of the phrase, not a
# use of it. The change-history pattern allows such a quoted mention.
